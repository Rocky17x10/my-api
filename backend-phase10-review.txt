import {
  Entity,
  PrimaryGeneratedColumn,
  Column,
  ManyToOne,
  JoinColumn,
  CreateDateColumn,
  UpdateDateColumn,
} from 'typeorm';

import { User } from '../users/user.entity';

export enum AssetRequestStatus {
  PENDING = 'pending',
  APPROVED = 'approved',
  REJECTED = 'rejected',
}

@Entity('asset_requests')
export class AssetRequest {
  @PrimaryGeneratedColumn()
  id: number;

  @ManyToOne(() => User, { nullable: false, onDelete: 'RESTRICT' })
  @JoinColumn({ name: 'requested_by' })
  requestedBy: User;

  @Column({ type: 'varchar', length: 150 })
  itemName: string;

  @Column({ type: 'varchar', length: 100 })
  category: string;

  @Column({ type: 'text' })
  reason: string;

  @Column({
    type: 'enum',
    enum: AssetRequestStatus,
    default: AssetRequestStatus.PENDING,
  })
  status: AssetRequestStatus;

  @ManyToOne(() => User, { nullable: true, onDelete: 'SET NULL' })
  @JoinColumn({ name: 'reviewed_by' })
  reviewedBy: User | null;

  @Column({ type: 'text', nullable: true })
  adminNote: string | null;

  @CreateDateColumn()
  createdAt: Date;

  @UpdateDateColumn()
  updatedAt: Date;
}

import {
  Injectable,
  BadRequestException,
  NotFoundException,
} from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';

import {
  AssetRequest,
  AssetRequestStatus,
} from './asset-request.entity';
import { User } from '../users/user.entity';

@Injectable()
export class AssetRequestsService {
  constructor(
    @InjectRepository(AssetRequest)
    private readonly requestRepository: Repository<AssetRequest>,

    @InjectRepository(User)
    private readonly userRepository: Repository<User>,
  ) {}

  async create(
    userId: number,
    data: { itemName: string; category: string; reason: string },
  ) {
    const itemName = data.itemName?.trim();
    const category = data.category?.trim();
    const reason = data.reason?.trim();

    if (!itemName || !category || !reason) {
      throw new BadRequestException(
        'Item name, category and reason are required.',
      );
    }

    const user = await this.userRepository.findOneBy({ id: userId });

    if (!user) {
      throw new NotFoundException('User not found.');
    }

    const request = this.requestRepository.create({
      requestedBy: user,
      itemName,
      category,
      reason,
      status: AssetRequestStatus.PENDING,
    });

    const saved = await this.requestRepository.save(request);

    return this.requestRepository.findOne({
      where: { id: saved.id },
      relations: { requestedBy: true, reviewedBy: true },
    });
  }

  async findMyRequests(userId: number) {
    return this.requestRepository.find({
      where: { requestedBy: { id: userId } },
      relations: { requestedBy: true, reviewedBy: true },
      order: { createdAt: 'DESC' },
    });
  }

  async findAll() {
    return this.requestRepository.find({
      relations: { requestedBy: true, reviewedBy: true },
      order: { createdAt: 'DESC' },
    });
  }

  async review(
    requestId: number,
    reviewerId: number,
    status: string,
    adminNote?: string,
  ) {
    if (
      status !== AssetRequestStatus.APPROVED &&
      status !== AssetRequestStatus.REJECTED
    ) {
      throw new BadRequestException(
        'Status must be approved or rejected.',
      );
    }

    const request = await this.requestRepository.findOne({
      where: { id: requestId },
      relations: { requestedBy: true, reviewedBy: true },
    });

    if (!request) {
      throw new NotFoundException('Asset request not found.');
    }

    if (request.status !== AssetRequestStatus.PENDING) {
      throw new BadRequestException(
        'This request has already been reviewed.',
      );
    }

    const reviewer = await this.userRepository.findOneBy({
      id: reviewerId,
    });

    if (!reviewer) {
      throw new NotFoundException('Reviewing user not found.');
    }

    request.status = status as AssetRequestStatus;
    request.reviewedBy = reviewer;
    request.adminNote = adminNote?.trim() || null;

    await this.requestRepository.save(request);

    return this.requestRepository.findOne({
      where: { id: requestId },
      relations: { requestedBy: true, reviewedBy: true },
    });
  }
}

import {
  Injectable,
  BadRequestException,
  NotFoundException,
  ForbiddenException,
  ConflictException,
} from '@nestjs/common';

import { InjectRepository } from '@nestjs/typeorm';
import {
  Repository,
  IsNull,
  EntityManager,
} from 'typeorm';

import { AssetReturn } from '../assets/asset-return.entity';
import { Asset } from '../assets/asset.entity';
import { AssetAssignmentHistory } from '../assets/asset-assignment-history.entity';
import { User } from '../users/user.entity';

type ReviewReturnData = {
  status: 'approved' | 'rejected';
  condition?: 'good' | 'damaged' | 'needs_maintenance';
  adminNote?: string;
};

@Injectable()
export class AssetReturnsService {
  constructor(
    @InjectRepository(AssetReturn)
    private readonly assetReturnRepository: Repository<AssetReturn>,

    @InjectRepository(Asset)
    private readonly assetRepository: Repository<Asset>,

    @InjectRepository(AssetAssignmentHistory)
    private readonly historyRepository: Repository<AssetAssignmentHistory>,

    @InjectRepository(User)
    private readonly userRepository: Repository<User>,
  ) {}

  // EMPLOYEE: Submit a return request
  async create(
    userId: number,
    assetId: number,
    employeeNote?: string,
  ) {
    if (!Number.isInteger(userId) || userId <= 0) {
      throw new BadRequestException('Invalid user ID');
    }

    if (!Number.isInteger(assetId) || assetId <= 0) {
      throw new BadRequestException('Invalid asset ID');
    }

    if (
      employeeNote !== undefined &&
      typeof employeeNote !== 'string'
    ) {
      throw new BadRequestException('Invalid employee note');
    }

    return this.assetReturnRepository.manager.transaction(
      async (manager: EntityManager) => {
        const assetRepo = manager.getRepository(Asset);
        const historyRepo = manager.getRepository(AssetAssignmentHistory);
        const returnRepo = manager.getRepository(AssetReturn);
        const userRepo = manager.getRepository(User);

        const asset = await assetRepo.findOne({
          where: { id: assetId },
          relations: { assignedTo: true },
          lock: { mode: 'pessimistic_write' },
        });

        if (!asset) {
          throw new NotFoundException('Asset not found');
        }

        if (
          asset.status !== 'assigned' ||
          !asset.assignedTo ||
          asset.assignedTo.id !== userId
        ) {
          throw new ForbiddenException(
            'You can only request the return of an asset assigned to you',
          );
        }

        const history = await historyRepo.findOne({
          where: {
            asset: { id: assetId },
            assignedTo: { id: userId },
            returnedAt: IsNull(),
          },
          order: { id: 'DESC' },
          lock: { mode: 'pessimistic_write' },
        });

        if (!history) {
          throw new NotFoundException(
            'Active assignment history not found',
          );
        }

        const existingRequest = await returnRepo.findOne({
          where: {
            assignmentHistory: { id: history.id },
            status: 'pending',
          },
        });

        if (existingRequest) {
          throw new ConflictException(
            'A return request is already pending for this asset',
          );
        }

        const employee = await userRepo.findOne({
          where: { id: userId },
        });

        if (!employee) {
          throw new NotFoundException('Employee not found');
        }

        const returnRequest = returnRepo.create({
          asset,
          assignmentHistory: history,
          requestedBy: employee,
          reviewedBy: null,
          status: 'pending',
          condition: null,
          employeeNote: employeeNote?.trim() || null,
          adminNote: null,
          reviewedAt: null,
          confirmedAt: null,
        });

        return returnRepo.save(returnRequest);
      },
    );
  }

  // EMPLOYEE: View own return requests
  async findMyReturns(userId: number) {
    return this.assetReturnRepository.find({
      where: {
        requestedBy: { id: userId },
      },
      relations: {
        asset: true,
        assignmentHistory: true,
        requestedBy: true,
        reviewedBy: true,
      },
      order: {
        requestedAt: 'DESC',
      },
    });
  }

  // ADMIN: View all return requests
  async findAllReturns() {
    return this.assetReturnRepository.find({
      relations: {
        asset: true,
        assignmentHistory: true,
        requestedBy: true,
        reviewedBy: true,
      },
      order: {
        requestedAt: 'DESC',
      },
    });
  }

  // ADMIN: Approve or reject a return request
  async review(
    returnId: number,
    adminId: number,
    data: ReviewReturnData,
  ) {
    if (!Number.isInteger(returnId) || returnId <= 0) {
      throw new BadRequestException('Invalid return request ID');
    }

    if (!Number.isInteger(adminId) || adminId <= 0) {
      throw new BadRequestException('Invalid admin ID');
    }

    if (
      !data ||
      !['approved', 'rejected'].includes(data.status)
    ) {
      throw new BadRequestException(
        'Status must be approved or rejected',
      );
    }

    if (
      data.adminNote !== undefined &&
      typeof data.adminNote !== 'string'
    ) {
      throw new BadRequestException('Invalid admin note');
    }

    const validConditions = [
      'good',
      'damaged',
      'needs_maintenance',
    ];

    if (
      data.status === 'approved' &&
      !validConditions.includes(data.condition as string)
    ) {
      throw new BadRequestException(
        'Please select a valid equipment condition',
      );
    }

    if (
      data.condition !== undefined &&
      !validConditions.includes(data.condition)
    ) {
      throw new BadRequestException('Invalid equipment condition');
    }

    return this.assetReturnRepository.manager.transaction(
      async (manager: EntityManager) => {
        const returnRepo = manager.getRepository(AssetReturn);
        const assetRepo = manager.getRepository(Asset);
        const historyRepo = manager.getRepository(AssetAssignmentHistory);
        const userRepo = manager.getRepository(User);

        // Lock the return request
        const lockedRequest = await returnRepo.findOne({
          where: { id: returnId },
          lock: { mode: 'pessimistic_write' },
        });

        if (!lockedRequest) {
          throw new NotFoundException('Return request not found');
        }

        if (lockedRequest.status !== 'pending') {
          throw new ConflictException(
            'This return request has already been reviewed',
          );
        }

        // Load the request's related records
        const returnRequest = await returnRepo.findOne({
          where: { id: returnId },
          relations: {
            asset: true,
            assignmentHistory: true,
            requestedBy: true,
          },
        });

        if (
          !returnRequest ||
          !returnRequest.asset ||
          !returnRequest.assignmentHistory ||
          !returnRequest.requestedBy
        ) {
          throw new NotFoundException(
            'Return request information is incomplete',
          );
        }

        const admin = await userRepo.findOne({
          where: { id: adminId },
        });

        if (!admin) {
          throw new NotFoundException('Admin not found');
        }

        // Lock the asset
        const asset = await assetRepo.findOne({
          where: { id: returnRequest.asset.id },
          relations: { assignedTo: true },
          lock: { mode: 'pessimistic_write' },
        });

        if (!asset) {
          throw new NotFoundException('Asset not found');
        }

        // Lock the assignment history
        const history = await historyRepo.findOne({
          where: { id: returnRequest.assignmentHistory.id },
          lock: { mode: 'pessimistic_write' },
        });

        if (!history) {
          throw new NotFoundException('Assignment history not found');
        }

        const now = new Date();

        if (data.status === 'approved') {
          if (
            asset.status !== 'assigned' ||
            !asset.assignedTo ||
            asset.assignedTo.id !== returnRequest.requestedBy.id ||
            history.returnedAt !== null
          ) {
            throw new ConflictException(
              'The asset assignment has changed. This return cannot be approved.',
            );
          }

          // Close assignment history
          history.returnedAt = now;
          await historyRepo.save(history);

          // Update asset
          asset.assignedTo = null as any;

          asset.status =
            data.condition === 'good'
              ? 'available'
              : 'maintenance';

          await assetRepo.save(asset);

          returnRequest.condition = data.condition!;
          returnRequest.confirmedAt = now;
        } else {
          // Rejection leaves the assignment unchanged
          returnRequest.condition = null;
          returnRequest.confirmedAt = null;
        }

        returnRequest.status = data.status;
        returnRequest.reviewedBy = admin;
        returnRequest.reviewedAt = now;
        returnRequest.adminNote =
          data.adminNote?.trim() || null;

        await returnRepo.save(returnRequest);

        return returnRepo.findOne({
          where: { id: returnId },
          relations: {
            asset: true,
            assignmentHistory: true,
            requestedBy: true,
            reviewedBy: true,
          },
        });
      },
    );
  }
}

import {
  Controller,
  Get,
  Post,
  Patch,
  Body,
  Param,
  ParseIntPipe,
  UseGuards,
  Req,
  ForbiddenException,
} from '@nestjs/common';

import type { Request } from 'express';

import { AssetsService } from './assets.service';

import { JwtAuthGuard } from '../auth/jwt-auth.guard';
import { RolesGuard } from '../auth/roles.guard';
import { Roles } from '../auth/roles.decorator';

interface AuthenticatedRequest extends Request {
  user: {
    userId: number;
    email: string;
    role: 'super_admin' | 'admin' | 'user';
  };
}

@Controller('assets')
@UseGuards(JwtAuthGuard, RolesGuard)
export class AssetsController {
  constructor(
    private readonly assetsService: AssetsService,
  ) {}

  // Get all assets
  @Get()
  findAll(@Req() req: AuthenticatedRequest) {
    const { userId, role } = req.user;

    if (role === 'super_admin' || role === 'admin') {
      return this.assetsService.findAll();
    }

    return this.assetsService.findAssignedToUser(userId);
  }

  // Get complete history of a specific asset
  // Only Admin and Super Admin can access
  @Get(':id/history')
  @Roles('super_admin', 'admin')
  findAssetHistory(
    @Param('id', ParseIntPipe) id: number,
  ) {
    return this.assetsService.findAssetHistory(id);
  }

  // Get assignment history of a specific employee
  // Employees can view only their own history
  @Get('history/employee/:employeeId')
  findEmployeeHistory(
    @Param('employeeId', ParseIntPipe) employeeId: number,
    @Req() req: AuthenticatedRequest,
  ) {
    const { userId, role } = req.user;

    if (
      role === 'user' &&
      userId !== employeeId
    ) {
      throw new ForbiddenException(
        'You can only view your own assignment history.',
      );
    }

    return this.assetsService.findEmployeeHistory(
      employeeId,
    );
  }

  // Create an asset
  @Post()
  @Roles('super_admin', 'admin')
  create(@Body() assetData: any) {
    return this.assetsService.create(assetData);
  }

  // Update an asset
  @Patch(':id')
  @Roles('super_admin', 'admin')
  update(
    @Param('id', ParseIntPipe) id: number,
    @Body() assetData: any,
  ) {
    return this.assetsService.update(id, assetData);
  }

  // Assign an asset and record the administrator
  @Post(':id/assign')
  @Roles('super_admin', 'admin')
  assignAsset(
    @Param('id', ParseIntPipe) id: number,
    @Body('userId') userId: number,
    @Req() req: AuthenticatedRequest,
  ) {
    return this.assetsService.assignAsset(
      id,
      userId,
      req.user.userId,
    );
  }

  // Unassign an asset and record its return date
  @Patch(':id/unassign')
  @Roles('super_admin', 'admin')
  unassignAsset(
    @Param('id', ParseIntPipe) id: number,
  ) {
    return this.assetsService.unassignAsset(id);
  }
}

import {
  Injectable,
  NotFoundException,
  BadRequestException,
  ConflictException,
} from '@nestjs/common';

import { InjectRepository } from '@nestjs/typeorm';
import { Repository, IsNull } from 'typeorm';

import { Asset } from './asset.entity';
import { AssetAssignmentHistory } from './asset-assignment-history.entity';
import { User } from '../users/user.entity';

@Injectable()
export class AssetsService {
  constructor(
    @InjectRepository(Asset)
    private readonly assetsRepository: Repository<Asset>,

    @InjectRepository(User)
    private readonly usersRepository: Repository<User>,

    @InjectRepository(AssetAssignmentHistory)
    private readonly historyRepository: Repository<AssetAssignmentHistory>,
  ) {}

  // Get all assets
  async findAll(): Promise<Asset[]> {
    return this.assetsRepository.find({
      relations: {
        assignedTo: true,
      },
      order: {
        id: 'DESC',
      },
    });
  }

  // Get assets assigned to a specific employee
  async findAssignedToUser(userId: number): Promise<Asset[]> {
    return this.assetsRepository.find({
      where: {
        assignedTo: {
          id: userId,
        },
      },
      relations: {
        assignedTo: true,
      },
      order: {
        id: 'DESC',
      },
    });
  }

  // Get complete assignment history of an asset
  async findAssetHistory(
    assetId: number,
  ): Promise<AssetAssignmentHistory[]> {
    const asset = await this.assetsRepository.findOneBy({
      id: assetId,
    });

    if (!asset) {
      throw new NotFoundException(
        `Asset with ID ${assetId} not found.`,
      );
    }

    return this.historyRepository.find({
      where: {
        asset: {
          id: assetId,
        },
      },
      relations: {
        asset: true,
        assignedTo: true,
        assignedBy: true,
      },
      order: {
        id: 'DESC',
      },
    });
  }

  // Get complete assignment history of an employee
  async findEmployeeHistory(
    employeeId: number,
  ): Promise<AssetAssignmentHistory[]> {
    const employee = await this.usersRepository.findOneBy({
      id: employeeId,
    });

    if (!employee) {
      throw new NotFoundException(
        `Employee with ID ${employeeId} not found.`,
      );
    }

    return this.historyRepository.find({
      where: {
        assignedTo: {
          id: employeeId,
        },
      },
      relations: {
        asset: true,
        assignedTo: true,
        assignedBy: true,
      },
      order: {
        id: 'DESC',
      },
    });
  }

  // Create an asset
  async create(assetData: any): Promise<Asset> {
    // Assignment must use the dedicated Assign API
    if (
      Object.prototype.hasOwnProperty.call(
        assetData,
        'assignedTo',
      )
    ) {
      throw new BadRequestException(
        'Assets cannot be assigned during creation. Use the Assign API.',
      );
    }

    const status = assetData.status ?? 'available';

    if (status === 'assigned') {
      throw new BadRequestException(
        'Assets cannot be created with Assigned status. Use the Assign API.',
      );
    }

    if (
      !['available', 'maintenance', 'retired'].includes(status)
    ) {
      throw new BadRequestException(
        'Invalid asset status.',
      );
    }

    const asset = new Asset();

    Object.assign(asset, assetData);

    asset.status = status;
    asset.assignedTo = null;

    return this.assetsRepository.save(asset);
  }

  // Update an asset
  async update(
    id: number,
    assetData: any,
  ): Promise<Asset> {
    const asset = await this.assetsRepository.findOne({
      where: { id },
      relations: {
        assignedTo: true,
      },
    });

    if (!asset) {
      throw new NotFoundException(
        `Asset with ID ${id} not found.`,
      );
    }

    // Assignment changes must use the dedicated Assign/Unassign APIs
    if (
      Object.prototype.hasOwnProperty.call(
        assetData,
        'assignedTo',
      )
    ) {
      throw new BadRequestException(
        'Assigned employee cannot be changed through asset editing. Use the Assign or Unassign API.',
      );
    }

    // Prevent directly setting Assigned status
    if (assetData.status === 'assigned') {
      throw new BadRequestException(
        'Assigned status can only be set through the Assign API.',
      );
    }

    // Validate a requested status change
    if (
      Object.prototype.hasOwnProperty.call(
        assetData,
        'status',
      )
    ) {
      if (
        !['available', 'maintenance', 'retired'].includes(
          assetData.status,
        )
      ) {
        throw new BadRequestException(
          'Invalid asset status.',
        );
      }

      if (asset.assignedTo || asset.status === 'assigned') {
        throw new BadRequestException(
          'Please unassign this asset before changing its status.',
        );
      }
    }

    // Preserve existing asset assignment information
    const { assignedTo, ...otherAssetData } = assetData;

    Object.assign(asset, otherAssetData);

    return this.assetsRepository.save(asset);
  }

  // Assign an available asset and create its history record
  async assignAsset(
    assetId: number,
    userId: number | string,
    adminId: number,
  ): Promise<Asset> {
    const employeeId = Number(userId);

    if (
      !Number.isInteger(employeeId) ||
      employeeId <= 0
    ) {
      throw new BadRequestException(
        'Invalid employee ID.',
      );
    }

    if (
      !Number.isInteger(adminId) ||
      adminId <= 0
    ) {
      throw new BadRequestException(
        'Invalid administrator ID.',
      );
    }

    return this.assetsRepository.manager.transaction(
      async (manager) => {
        const asset = await manager.findOne(Asset, {
          where: { id: assetId },
          relations: {
            assignedTo: true,
          },
          lock: {
            mode: 'pessimistic_write',
          },
        });

        if (!asset) {
          throw new NotFoundException(
            `Asset with ID ${assetId} not found.`,
          );
        }

        if (
          asset.status !== 'available' ||
          asset.assignedTo !== null
        ) {
          throw new ConflictException(
            'This asset is not available for assignment.',
          );
        }

        const employee = await manager.findOneBy(
          User,
          { id: employeeId },
        );

        if (!employee) {
          throw new NotFoundException(
            `Employee with ID ${employeeId} not found.`,
          );
        }

        const admin = await manager.findOneBy(
          User,
          { id: adminId },
        );

        if (!admin) {
          throw new NotFoundException(
            `Administrator with ID ${adminId} not found.`,
          );
        }

        asset.assignedTo = employee;
        asset.status = 'assigned';

        await manager.save(Asset, asset);

        const history = manager.create(
          AssetAssignmentHistory,
          {
            asset,
            assignedTo: employee,
            assignedBy: admin,
            returnedAt: null,
          },
        );

        await manager.save(
          AssetAssignmentHistory,
          history,
        );

        return asset;
      },
    );
  }

  // Unassign an asset and record its return date
  async unassignAsset(
    assetId: number,
  ): Promise<Asset> {
    return this.assetsRepository.manager.transaction(
      async (manager) => {
        const asset = await manager.findOne(Asset, {
          where: { id: assetId },
          relations: {
            assignedTo: true,
          },
          lock: {
            mode: 'pessimistic_write',
          },
        });

        if (!asset) {
          throw new NotFoundException(
            `Asset with ID ${assetId} not found.`,
          );
        }

        if (
          asset.status !== 'assigned' ||
          !asset.assignedTo
        ) {
          throw new BadRequestException(
            'This asset is not currently assigned.',
          );
        }

        const history = await manager.findOne(
          AssetAssignmentHistory,
          {
            where: {
              asset: {
                id: assetId,
              },
              returnedAt: IsNull(),
            },
            order: {
              id: 'DESC',
            },
            lock: {
              mode: 'pessimistic_write',
            },
          },
        );

        if (!history) {
          throw new BadRequestException(
            'No active assignment history found for this asset.',
          );
        }

        history.returnedAt = new Date();

        await manager.save(
          AssetAssignmentHistory,
          history,
        );

        asset.assignedTo = null;
        asset.status = 'available';

        return manager.save(Asset, asset);
      },
    );
  }
}

import {
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import * as bcrypt from 'bcrypt';
import { UsersService } from '../users/users.service';

@Injectable()
export class AuthService {
  constructor(
    private readonly usersService: UsersService,
    private readonly jwtService: JwtService,
  ) {}

  async login(email: string, password: string) {
    const user = await this.usersService.findByEmail(email);

    if (!user || !user.password) {
      throw new UnauthorizedException(
        'Invalid email or password',
      );
    }

    const isPasswordValid = await bcrypt.compare(
      password,
      user.password,
    );

    if (!isPasswordValid) {
      throw new UnauthorizedException(
        'Invalid email or password',
      );
    }

    const payload = {
      sub: user.id,
      email: user.email,
      role: user.role,
    };

    return {
      access_token: await this.jwtService.signAsync(payload),
      user: {
        id: user.id,
        name: user.name,
        email: user.email,
        role: user.role,
        department: user.department,
        designation: user.designation,
      },
    };
  }
}

import { Injectable } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { PassportStrategy } from '@nestjs/passport';
import { ExtractJwt, Strategy } from 'passport-jwt';
import { UsersService } from '../users/users.service';

interface JwtPayload {
  sub: number;
  email: string;
}

@Injectable()
export class JwtStrategy extends PassportStrategy(Strategy) {
  constructor(
    configService: ConfigService,
    private readonly usersService: UsersService,
  ) {
    super({
      jwtFromRequest: ExtractJwt.fromAuthHeaderAsBearerToken(),
      ignoreExpiration: false,
      secretOrKey: configService.getOrThrow<string>('JWT_SECRET'),
    });
  }

  async validate(payload: JwtPayload) {
    const user = await this.usersService.findOne(payload.sub);

    return {
      userId: user.id,
      email: user.email,
      name: user.name,
      role: user.role,
      department: user.department,
      designation: user.designation,
    };
  }
}

===== src/auth/roles.guard.ts =====

import {
  Injectable,
  CanActivate,
  ExecutionContext,
} from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { ROLES_KEY } from './roles.decorator';
import { UserRole } from '../users/user.entity';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private readonly reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const requiredRoles = this.reflector.getAllAndOverride<
      UserRole[]
    >(ROLES_KEY, [
      context.getHandler(),
      context.getClass(),
    ]);

    if (!requiredRoles || requiredRoles.length === 0) {
      return true;
    }

    const request = context.switchToHttp().getRequest();

    const user = request.user as {
      role?: UserRole;
    };

    return !!user?.role && requiredRoles.includes(user.role);
  }
}

import {
  Body,
  BadRequestException,
  Controller,
  Delete,
  Get,
  Param,
  ParseIntPipe,
  Patch,
  Post,
  UseGuards,
  Req,
} from '@nestjs/common';

import type { Request } from 'express';

import { UsersService } from './users.service';
import { User } from './user.entity';
import type { UserRole } from './user.entity';

import { CreateUserDto } from './create-user.dto';
import { UpdateUserDto } from './update-user.dto';

import { JwtAuthGuard } from '../auth/jwt-auth.guard';
import { RolesGuard } from '../auth/roles.guard';
import { Roles } from '../auth/roles.decorator';

interface AuthenticatedRequest extends Request {
  user: {
    userId: number;
    email: string;
    role: UserRole;
  };
}

@Controller('users')
@UseGuards(JwtAuthGuard, RolesGuard)
@Roles('super_admin', 'admin')
export class UsersController {
  constructor(
    private readonly usersService: UsersService,
  ) {}

  // GET /users
  @Get()
  findAll(): Promise<User[]> {
    return this.usersService.findAll();
  }

  // GET /users/:id
  @Get(':id')
  findOne(
    @Param('id', ParseIntPipe) id: number,
  ): Promise<User> {
    return this.usersService.findOne(id);
  }

  // POST /users
  @Post()
  create(
    @Body() createUserDto: CreateUserDto,
  ): Promise<User> {
    return this.usersService.create(
      createUserDto.name,
      createUserDto.email,
      createUserDto.password,
      createUserDto.department,
      createUserDto.designation,
    );
  }

  // PATCH /users/:id
  @Patch(':id')
  update(
    @Param('id', ParseIntPipe) id: number,
    @Body() updateUserDto: UpdateUserDto,
  ): Promise<User> {
    if (
      updateUserDto.name === undefined &&
      updateUserDto.email === undefined &&
      updateUserDto.department === undefined &&
      updateUserDto.designation === undefined
    ) {
      throw new BadRequestException(
        'Provide at least one field to update',
      );
    }

    return this.usersService.update(
      id,
      updateUserDto.name,
      updateUserDto.email,
      updateUserDto.department,
      updateUserDto.designation,
    );
  }

  // PATCH /users/:id/role
  // Only Super Admin can promote or demote accounts
  @Patch(':id/role')
  @Roles('super_admin')
  changeRole(
    @Param('id', ParseIntPipe) id: number,
    @Body('role') role: string,
    @Req() req: AuthenticatedRequest,
  ): Promise<User> {
    if (role !== 'admin' && role !== 'user') {
      throw new BadRequestException(
        'Role must be either admin or user.',
      );
    }

    return this.usersService.changeRole(
      id,
      role as UserRole,
      req.user.userId,
    );
  }

  // DELETE /users/:id
  @Delete(':id')
  remove(
    @Param('id', ParseIntPipe) id: number,
  ) {
    return this.usersService.remove(id);
  }
}

===== src/users/users.service.ts =====

import {
  Injectable,
  NotFoundException,
  ForbiddenException,
  BadRequestException,
} from '@nestjs/common';

import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import * as bcrypt from 'bcrypt';

import { User, UserRole } from './user.entity';

@Injectable()
export class UsersService {
  constructor(
    @InjectRepository(User)
    private readonly usersRepository: Repository<User>,
  ) {}

  // Find a user by email for authentication
  async findByEmail(email: string): Promise<User | null> {
    return this.usersRepository
      .createQueryBuilder('user')
      .addSelect('user.password')
      .where('user.email = :email', { email })
      .getOne();
  }

  // GET /users
  findAll(): Promise<User[]> {
    return this.usersRepository.find({
      order: { id: 'DESC' },
    });
  }

  // GET /users/:id
  async findOne(id: number): Promise<User> {
    const user = await this.usersRepository.findOneBy({ id });

    if (!user) {
      throw new NotFoundException('User not found');
    }

    return user;
  }

  // POST /users
  async create(
    name: string,
    email: string,
    password: string,
    department?: string,
    designation?: string,
  ): Promise<User> {
    const hashedPassword = await bcrypt.hash(password, 12);

    const user = this.usersRepository.create({
      name,
      email,
      password: hashedPassword,
      department: department || null,
      designation: designation || null,
    });

    const savedUser = await this.usersRepository.save(user);

    const { password: _password, ...safeUser } = savedUser;

    return safeUser as User;
  }

  // PATCH /users/:id
  async update(
    id: number,
    name?: string,
    email?: string,
    department?: string,
    designation?: string,
  ): Promise<User> {
    const user = await this.findOne(id);

    // Protect the Super Admin account
    if (user.role === 'super_admin') {
      throw new ForbiddenException(
        'The Super Admin account cannot be modified.',
      );
    }

    if (name !== undefined) {
      user.name = name;
    }

    if (email !== undefined) {
      user.email = email;
    }

    if (department !== undefined) {
      user.department = department || null;
    }

    if (designation !== undefined) {
      user.designation = designation || null;
    }

    return this.usersRepository.save(user);
  }

  // PATCH /users/:id/role
  // Only Super Admin can change another account's role
  async changeRole(
    id: number,
    role: UserRole,
    actorId: number,
  ): Promise<User> {
    const user = await this.findOne(id);

    if (id === actorId) {
      throw new ForbiddenException(
        'You cannot change your own role.',
      );
    }

    if (user.role === 'super_admin') {
      throw new ForbiddenException(
        'The Super Admin role cannot be changed.',
      );
    }

    if (role !== 'admin' && role !== 'user') {
      throw new BadRequestException(
        'Role must be either admin or user.',
      );
    }

    user.role = role;

    return this.usersRepository.save(user);
  }

  // DELETE /users/:id
  async remove(id: number): Promise<{ message: string }> {
    const user = await this.findOne(id);

    // Never allow deletion of the Super Admin account
    if (user.role === 'super_admin') {
      throw new ForbiddenException(
        'The Super Admin account cannot be deleted.',
      );
    }

    await this.usersRepository.remove(user);

    return { message: 'User deleted successfully' };
  }
}
